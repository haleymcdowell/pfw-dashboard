// Create a dropdown menu of the county name.
viewof countyNameDropdown = Inputs.select(
  Array.from(new Set(recent_observations.map(d => d.Subregion))),
  {
    label: "Choose a county",
    unique: true, // Only include each location once.
    sort: true // Sort the options alphabetically.
    }
)
// Create a dropdown menu of the city name.
viewof cityNameDropdown = Inputs.select(
  Array.from(new Set(recent_observations.filter(d => d.Subregion === countyNameDropdown)
  .map(d => d.City))),
  {
    label: "Choose a city",
    unique: true, // Only include each location once.
    sort: true // Sort the options alphabetically.
    }
)
// Create a dropdown menu of the location name.
viewof selected_location = Inputs.select(
  Array.from(new Set(recent_observations.filter(d => d.City === cityNameDropdown)
  .map(d => d.locName))),
  {
    label: "Choose a location",
    unique: true, // Only include each location once.
    sort: true // Sort the options alphabetically.
    }
)
filtered_observations = recent_observations.filter(d => d.locName === selected_location)

// Create a horizontal bar graph using Observable Plot.
Plot.plot({
  title: "Recent Observations for " + selected_location,
  marginLeft: 150,
  height: filtered_observations.length * 25 + 100,
  width: undefined,
  x: {
    label: "Number Observed"
  },
  y: {
    label: "Species Common Name"
  },
  marks: [
    Plot.barX(filtered_observations, {
      x: "howMany",
      y: "comName",
      sort: { y: "x", reverse: true }, // Sort data largest to smallest.
      fill: "#888"
    }),
    // Add data labels.
   Plot.text(filtered_observations, {
      x: "howMany",
      y: "comName",
      text: d => d.howMany, 
      dx: -15,
      dy: 0,
      fill: "#fff",
      textAnchor: "start",
    }),
  ]
})
